/* API helpers */

async function apiRequest(url, options = {}) {
  const response = await fetch(url, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...options.headers,
    },
  })

  const text = await response.text()
  let data = null

  try {
    data = text ? JSON.parse(text) : null
  } catch {
    data = text
  }

  if (!response.ok) {
    throw new Error(
      typeof data === 'object' && data !== null
        ? data.message || data.error || `HTTP error ${response.status}`
        : `HTTP error ${response.status}: ${data || response.statusText}`
    )
  }

  return data
}

function normalizeOrganization(org) {
  const currencyValue =
    org.currency ??
    org.reporting_currency ??
    org.currency_code ??
    org.reportingCurrency ??
    'INR'

  const typeValue =
    org.type ??
    org.organization_type ??
    org.org_type ??
    ''

  const licenseValue =
    org.license ??
    org.license_organization ??
    org.licenseOrganization ??
    false

  return {
    ...org,

    // Keep the database ID for PUT and DELETE.
    id: org.id ?? org.organization_id ?? org.organizationId ?? null,

    code: String(
      org.code ?? org.organization_code ?? org.organizationCode ?? ''
    ),

    name: String(
      org.name ?? org.organization_name ?? org.organizationName ?? ''
    ),

    currency:
      typeof currencyValue === 'object'
        ? currencyValue.currency_code ?? currencyValue.code ?? 'INR'
        : String(currencyValue),

    status: String(org.status ?? org.organization_status ?? 'Active'),

    license:
      licenseValue === true ||
      licenseValue === 1 ||
      licenseValue === '1' ||
      licenseValue === 'true',

    type:
      typeof typeValue === 'object'
        ? typeValue.name ?? typeValue.type_name ?? ''
        : String(typeValue),

    industry:
      typeof org.industry === 'object'
        ? org.industry.name ?? org.industry.industry_name ?? ''
        : org.industry ?? org.industry_name ?? '',

    registrationNumber:
      org.registrationNumber ?? org.registration_number ?? '',

    incorporationDate: String(
      org.incorporationDate ??
      org.incorporation_date ??
      org.date_of_incorporation ??
      ''
    ).slice(0, 10),

    taxId: org.taxId ?? org.tax_id ?? '',

    address:
      org.address ?? org.registered_address ?? '',

    country:
      typeof org.country === 'object'
        ? org.country.country_name ?? org.country.name ?? ''
        : org.country ?? org.country_name ?? '',

    state:
      typeof org.state === 'object'
        ? org.state.state_name ?? org.state.name ?? ''
        : org.state ?? org.state_name ?? '',

    city: org.city ?? '',

    pincode:
      org.pincode ?? org.postal_code ?? org.zip_code ?? '',

    phone: org.phone ?? org.phone_number ?? '',

    email: org.email ?? '',

    website: org.website ?? '',
  }
}

function getOrganizationIdentifier(org) {
  // Prefer the database primary key.
  // Fall back to organization code if the API uses codes in its routes.
  return org.id ?? org.code
}

function getOrganizationArray(result) {
  if (Array.isArray(result)) return result
  if (Array.isArray(result?.data)) return result.data
  if (Array.isArray(result?.organizations)) return result.organizations
  if (Array.isArray(result?.results)) return result.results

  return null
}

      console.log('Payload:', payload)

                </div>

                <div className="org-status-field">
                  <label htmlFor="organization-status">Status</label>
                  <select
                    id="organization-status"
                    value={statusFilter}
                    onChange={(event) => setStatusFilter(event.target.value)}
                  >
                    <option value="">All Statuses</option>
                    <option value="Active">Active</option>
                    <option value="Inactive">Inactive</option>
                  </select>
                </div>

                <button
                  type="button"
                  className="primary-btn"
                  onClick={runSearch}
                >
                  Search
                </button>

                <button
                  type="button"
                  className="secondary-btn"
                  onClick={clearSearch}
                >
                  Clear
                </button>
              </div>

              <div className="org-table-wrapper">
                <table className="org-table">
                  <thead>
                    <tr>
                      <th>Select</th>
                      <th>Organization Code</th>
                      <th>Organization Name</th>
                      <th>Reporting Currency</th>
                      <th>Status</th>
                    </tr>
                  </thead>

                  <tbody>
                    {visibleOrganizations.map((org) => (
                      <tr
                        key={org.code}
                        className={selectedCode === org.code ? 'selected' : ''}
                        onClick={() => setSelectedCode(org.code)}
                      >
                        <td>
                          <input
                            type="radio"
                            name="selectedOrganization"
                            aria-label={`Select ${org.name}`}
                            checked={selectedCode === org.code}
                            onChange={() => setSelectedCode(org.code)}
                            onClick={(event) => event.stopPropagation()}
                          />
                        </td>
                        <td>{org.code}</td>
                        <td>{org.name}</td>
                        <td>{org.currency || '—'}</td>
                        <td>
                          <span
                            className={
                              org.status === 'Active'
                                ? 'status-active'
                                : 'status-inactive'
                            }
                          >
                            {org.status}
                          </span>
                        </td>
                      </tr>
                    ))}

                    {visibleOrganizations.length === 0 && (
                      <tr>
                        <td colSpan={5} className="empty-table">
                          No organizations found. Try different search criteria.
                        </td>
                      </tr>
                    )}
                  </tbody>
                </table>
              </div>

              <div className="org-pagination">
                <label>
                  Lines per Page
                  <select
                    value={rowsPerPage}
                    onChange={(event) => {
                      setRowsPerPage(Number(event.target.value))
                      setPage(1)
                    }}
                  >
                    <option value={5}>5</option>
                    <option value={10}>10</option>
                    <option value={20}>20</option>
                    <option value={50}>50</option>
                  </select>
                </label>

                <span>
                  {filteredOrganizations.length === 0
                    ? '0 records'
                    : `${(page - 1) * rowsPerPage + 1}–${Math.min(
                        page * rowsPerPage,
                        filteredOrganizations.length
                      )} of ${filteredOrganizations.length}`}
                </span>

                <button
                  type="button"
                  disabled={page <= 1}
                  onClick={() => setPage((previous) => previous - 1)}
                >
                  Previous
                </button>

                <span>
                  Page {page} of {totalPages}
                </span>

                <button
                  type="button"
                  disabled={page >= totalPages}
                  onClick={() => setPage((previous) => previous + 1)}
                >
                  Next
                </button>
              </div>

              <div className="org-bottom-actions">
                <button
                  type="button"
                  className="primary-btn"
                  onClick={openSelectedOrganization}
                >
                  View Selected
                </button>
                <button
                  type="button"
                  className="secondary-btn"
                  onClick={openEditScreen}
                >
                  Edit Selected
                </button>
                <button
                  type="button"
                  className="secondary-btn"
                  onClick={openAddScreen}
                >
                  + Add Organization
                </button>
              </div>
            </section>

            <section className="org-panel">
              <div className="org-section-heading">
                <h2>Selected Organization Summary</h2>
                <span>
                  {selectedOrganization ? 'Selected' : 'No selection'}
                </span>
              </div>

              {selectedOrganization ? (
                <div className="details-grid">
                  <div className="detail-item">
                    <span>Organization Code</span>
                    <strong>{selectedOrganization.code}</strong>
                  </div>
                  <div className="detail-item">
                    <span>Organization Name</span>
                    <strong>{selectedOrganization.name}</strong>
                  </div>
                  <div className="detail-item">
                    <span>Status</span>
                    <strong>
                      <span
                        className={
                          selectedOrganization.status === 'Active'
                            ? 'status-active'
                            : 'status-inactive'
                        }
                      >
                        {selectedOrganization.status}
                      </span>
                    </strong>
                  </div>
                </div>
              ) : (
                <p>Select an organization row to see its summary here.</p>
              )}
            </section>
          </>
        )}

        {screen === 'add' && (
          <section className="org-panel">
            <div className="org-section-heading">
              <h2>Organization Details</h2>
              <span>{editingCode ? 'Edit Details' : 'Main Details'}</span>
            </div>

            <form
              id="organization-form"
              className="org-form"
              onSubmit={saveOrganization}
              noValidate
            >
              <div className="form-grid">
                <FormField label="Organization Code *" error={errors.code}>
                  <input
                    name="code"
                    value={form.code}
                    onChange={updateField}
                    maxLength={30}
                    required
                    disabled={Boolean(editingCode)}
                    aria-invalid={Boolean(errors.code)}
                    placeholder="Enter organization code"
                  />
                </FormField>

                <FormField label="Organization Name *" error={errors.name}>
                  <input
                    name="name"
                    value={form.name}
                    onChange={updateField}
                    maxLength={150}
                    required
                    aria-invalid={Boolean(errors.name)}
                    placeholder="Enter organization name"
                  />
                </FormField>

                <FormField label="Status">
                  <select
                    name="status"
                    value={form.status}
                    onChange={updateField}
                  >
                    <option value="Active">Active</option>
                    <option value="Inactive">Inactive</option>
                  </select>
                </FormField>

                <FormField label="Organization Type">
                  <select name="type" value={form.type} onChange={updateField}>
                    <option value="">Select Type</option>
                    <option value="Head Office">Head Office</option>
                    <option value="Branch">Branch</option>
                    <option value="Subsidiary">Subsidiary</option>
                    <option value="Other">Other</option>
                  </select>
                </FormField>

                <FormField
                  label="Reporting Currency *"
                  error={errors.currency}
                >
                  <select
                    name="currency"
                    value={form.currency}
                    onChange={updateField}
                    required
                    aria-invalid={Boolean(errors.currency)}
                  >
                    <option value="INR">INR - Indian Rupee</option>
                    <option value="USD">USD - US Dollar</option>
                    <option value="EUR">EUR - Euro</option>
                    <option value="GBP">GBP - British Pound</option>
                  </select>
                </FormField>

                <FormField label="Industry">
                  <select
                    name="industry"
                    value={form.industry}
                    onChange={updateField}
                  >
                    <option value="">Select Industry</option>
                    <option value="Information Technology">
                      Information Technology
                    </option>
                    <option value="Manufacturing">Manufacturing</option>
                    <option value="Healthcare">Healthcare</option>
                    <option value="Education">Education</option>
                    <option value="Logistics">Logistics</option>
                    <option value="Other">Other</option>
                  </select>
                </FormField>

                <FormField label="Registration Number">
                  <input
                    name="registrationNumber"
                    value={form.registrationNumber}
                    onChange={updateField}
                  />
                </FormField>

                <FormField label="Date of Incorporation">
                  <input
                    type="date"
                    name="incorporationDate"
                    value={form.incorporationDate}
                    onChange={updateField}
                  />
                </FormField>

                <FormField label="Tax ID">
                  <input
                    name="taxId"
                    value={form.taxId}
                    onChange={updateField}
                  />
                </FormField>

                <FormField label="Country *" error={errors.country}>
                  <select
                    name="country"
                    value={form.country}
                    onChange={updateField}
                    required
                    aria-invalid={Boolean(errors.country)}
                  >
                    <option value="">Select Country</option>
                    {countries.map((country) => (
                      <option key={country} value={country}>
                        {country}
                      </option>
                    ))}
                  </select>
                </FormField>

                <FormField label="State / Province *" error={errors.state}>
                  <select
                    name="state"
                    value={form.state}
                    onChange={updateField}
                    required
                    disabled={!form.country}
                    aria-invalid={Boolean(errors.state)}
                  >
                    <option value="">Select State / Province</option>
                    {(countryStates[form.country] || []).map((stateName) => (
                      <option key={stateName} value={stateName}>
                        {stateName}
                      </option>
                    ))}
                  </select>
                </FormField>

                <FormField label="City">
                  <input
                    name="city"
                    value={form.city}
                    onChange={updateField}
                    placeholder="Enter city"
                  />
                </FormField>

                <FormField label="Pincode / Postal Code" error={errors.pincode}>
                  <input
                    name="pincode"
                    value={form.pincode}
                    onChange={updateField}
                    aria-invalid={Boolean(errors.pincode)}
                    placeholder="Enter postal code"
                  />
                </FormField>

                <FormField label="Phone" error={errors.phone}>
                  <input
                    type="tel"
                    name="phone"
                    value={form.phone}
                    onChange={updateField}
                    aria-invalid={Boolean(errors.phone)}
                    placeholder="Enter phone number"
                  />
                </FormField>

                <FormField label="Email" error={errors.email}>
                  <input
                    type="email"
                    name="email"
                    value={form.email}
                    onChange={updateField}
                    aria-invalid={Boolean(errors.email)}
                    placeholder="name@example.com"
                  />
                </FormField>

                <FormField label="Website">
                  <input
                    type="url"
                    name="website"
                    value={form.website}
                    onChange={updateField}
                    placeholder="https://example.com"
                  />
                </FormField>
              </div>

              <div className="license-row">
                <label>
                  <input
                    type="checkbox"
                    name="license"
                    checked={form.license}
                    onChange={updateField}
                  />
                  License Organization
                </label>
              </div>

              <FormField label="Registered Address">
                <textarea
                  name="address"
                  value={form.address}
                  onChange={updateField}
                  rows={3}
                  placeholder="Enter registered address"
                />
              </FormField>

              <div className="form-actions">
                <button
                  type="button"
                  className="secondary-btn"
                  onClick={goToList}
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="primary-btn"
                  disabled={loading}
                >
                  {loading
                    ? 'Please wait...'
                    : editingCode
                      ? 'Update Organization'
                      : 'Save Organization'}
                </button>
              </div>
            </form>
          </section>
        )}

        {screen === 'details' && viewOrganization && (
          <section className="org-panel">
            <div className="org-section-heading">
              <h2>Organization Details</h2>
              <span>{viewOrganization.status}</span>
            </div>

            <div className="details-grid">
              {Object.entries(viewOrganization)
                .filter(([key]) => key !== 'id')
                .map(([key, value]) => (
                  <div className="detail-item" key={key}>
                    <span>{displayLabel(key)}</span>
                    <strong>
                      {typeof value === 'boolean'
                        ? value
                          ? 'Yes'
                          : 'No'
                        : value || '—'}
                    </strong>
                  </div>
                ))}
            </div>

            <div className="form-actions">
              <button
                type="button"
                className="secondary-btn"
                onClick={goToList}
              >
                Back to List
              </button>
              <button
                type="button"
                className="primary-btn"
                onClick={() => {
                  setSelectedCode(viewOrganization.code)
                  setForm({ ...emptyForm, ...viewOrganization })
                  setEditingCode(viewOrganization.code)
                  setErrors({})
                  setScreen('add')
                }}
              >
                Edit Organization
              </button>
            </div>
          </section>
        )}
      </main>

      <footer className="app-footer">
        150CHF Organization Management System
      </footer>

      {showSelector && (
        <div
          className="modal-backdrop"
          role="presentation"
          onMouseDown={(event) => {
            if (event.target === event.currentTarget) {
              setShowSelector(false)
            }
          }}
        >
          <section
            className="organization-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="organization-modal-title"
          >
            <div className="modal-header">
              <h2 id="organization-modal-title">Select an Organization</h2>
              <button
                type="button"
                aria-label="Close organization selector"
                onClick={() => setShowSelector(false)}
              >
                ×
              </button>
            </div>

            <div className="modal-search-grid">
              <FormField label="Organization Code">
                <input
                  value={modalCode}
                  onChange={(event) => setModalCode(event.target.value)}
                  onKeyDown={(event) => {
                    if (event.key === 'Enter') runModalSearch()
                  }}
                  placeholder="Enter code"
                />
              </FormField>

              <FormField label="Organization Name">
                <input
                  value={modalName}
                  onChange={(event) => setModalName(event.target.value)}
                  onKeyDown={(event) => {
                    if (event.key === 'Enter') runModalSearch()
                  }}
                  placeholder="Enter organization name"
                />
              </FormField>

              <FormField label="Status">
                <select
                  value={modalStatus}
                  onChange={(event) => setModalStatus(event.target.value)}
                >
                  <option value="">All Statuses</option>
                  <option value="Active">Active</option>
                  <option value="Inactive">Inactive</option>
                </select>
              </FormField>
            </div>

            <div className="modal-actions">
              <button
                type="button"
                className="primary-btn"
                onClick={runModalSearch}
              >
                Search
              </button>
              <button
                type="button"
                className="secondary-btn"
                onClick={clearModalSearch}
              >
                Clear
              </button>
            </div>

            <div className="org-table-wrapper">
              <table className="org-table">
                <thead>
                  <tr>
                    <th>Select</th>
                    <th>Code</th>
                    <th>Organization Name</th>
                    <th>Status</th>
                  </tr>
                </thead>
                <tbody>
                  {visibleModalOrganizations.map((org) => (
                    <tr
                      key={org.code}
                      className={selectedCode === org.code ? 'selected' : ''}
                      onClick={() => setSelectedCode(org.code)}
                    >
                      <td>
                        <input
                          type="radio"
                          name="modalOrganization"
                          checked={selectedCode === org.code}
                          onChange={() => setSelectedCode(org.code)}
                          aria-label={`Select ${org.name}`}
                        />
                      </td>
                      <td>{org.code}</td>
                      <td>{org.name}</td>
                      <td>
                        <span
                          className={
                            org.status === 'Active'
                              ? 'status-active'
                              : 'status-inactive'
                          }
                        >
                          {org.status}
                        </span>
                      </td>
                    </tr>
                  ))}

                  {visibleModalOrganizations.length === 0 && (
                    <tr>
                      <td colSpan={4} className="empty-table">
                        No organizations found.
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>

            <div className="org-pagination">
              <label>
                Lines per Page
                <select
                  value={modalRowsPerPage}
                  onChange={(event) => {
                    setModalRowsPerPage(Number(event.target.value))
                    setModalPage(1)
                  }}
                >
                  <option value={5}>5</option>
                  <option value={10}>10</option>
                  <option value={20}>20</option>
                </select>
              </label>

              <span>
                {modalOrganizations.length === 0
                  ? '0 records'
                  : `${(modalPage - 1) * modalRowsPerPage + 1}–${Math.min(
                      modalPage * modalRowsPerPage,
                      modalOrganizations.length
                    )} of ${modalOrganizations.length}`}
              </span>

              <button
                type="button"
                disabled={modalPage <= 1}
                onClick={() => setModalPage((previous) => previous - 1)}
              >
                Previous
              </button>
              <span>
                Page {modalPage} of {modalTotalPages}
              </span>
              <button
                type="button"
                disabled={modalPage >= modalTotalPages}
                onClick={() => setModalPage((previous) => previous + 1)}
              >
                Next
              </button>
            </div>

            <div className="form-actions">
              <button
                type="button"
                className="secondary-btn"
                onClick={() => setShowSelector(false)}
              >
                Cancel
              </button>
              <button
                type="button"
                className="primary-btn"
                disabled={!selectedCode}
                onClick={() => chooseOrganization(selectedCode)}
              >
                Select Organization
              </button>
            </div>
          </section>
        </div>
      )}

      {showDeleteConfirm && (
        <div className="modal-backdrop">
          <section
            className="organization-modal delete-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="delete-title"
          >
            <div className="modal-header">
              <h2 id="delete-title">Confirm Delete</h2>
            </div>
            <p>
              Are you sure you want to delete{' '}
              <strong>{selectedOrganization?.name}</strong> (
              {selectedOrganization?.code})?
            </p>
            <div className="form-actions">
              <button
                type="button"
                className="secondary-btn"
                onClick={() => setShowDeleteConfirm(false)}
              >
                Cancel
              </button>
              <button
                type="button"
                className="primary-btn"
                onClick={confirmDelete}
                disabled={loading}
              >
                {loading ? 'Please wait...' : 'Delete'}
              </button>
            </div>
          </section>
        </div>
      )}
    </div>
  )
}

function FormField({ label, children, error }) {
  return (
    <div className="form-field">
      <label>{label}</label>
      {children}
      {error && (
        <small className="field-error" role="alert">
          {error}
        </small>
      )}
    </div>
  )
}

export default App
